#!/bin/bash
# Read existing debugfs views only. Never mounts debugfs or accesses regmap dumps.
set -euo pipefail
project=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
remote_command=$(cat <<'REMOTE'
sudo -- sh -c 'printf "\nPOLARIS_DMESG_BEGIN\n"; printf "[gpio]\n"; cat /sys/kernel/debug/gpio; printf "[touch-irq]\n"; cat /sys/kernel/debug/irq/irqs/147; printf "[regulators]\n"; cat /sys/kernel/debug/regulator/regulator_summary; printf "[clocks]\n"; cat /sys/kernel/debug/clk/clk_summary; printf "[rmi-sysfs]\n"; ls -la /sys/bus/rmi4/devices/; true'
REMOTE
)
ssh -F /dev/null -tt -S "$project/build/ssh/control" \
    -o StrictHostKeyChecking=yes \
    -o UserKnownHostsFile=<用户目录>/.ssh/known_hosts \
    -o ConnectTimeout=5 user@172.16.42.1 "$remote_command" \
    | python3 -B -u "$project/tools/redact-log.py" --auth-until-marker \
    | tee "$project/logs/device-debug-readonly-20261104.log"
