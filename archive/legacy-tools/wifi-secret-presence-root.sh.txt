#!/bin/sh
# Inspect availability only. Never emit secret data, including through errors.
set -eu
set +x
export LC_ALL=C PAGER=cat NM_CLI_PAGER=cat
test "$(nmcli -g GENERAL.CONNECTION device show wlan0)" = polaris-<2.4GHz热点>
value=$(nmcli --show-secrets -g 802-11-wireless-security.psk connection show polaris-<2.4GHz热点>)
if test -n "$value" && test "$value" != --; then
    printf 'Active Wi-Fi secret available in NetworkManager memory: yes\n'
else
    printf 'Active Wi-Fi secret available in NetworkManager memory: no\n'
fi
unset value
printf 'Host credential key already exists: '
if test -f /var/lib/systemd/credential.secret; then echo yes; else echo no; fi
printf 'Swap devices: '; awk 'END { print NR-1 }' /proc/swaps
printf 'Encrypted credential directory exists: '
if test -d /etc/credstore.encrypted; then echo yes; else echo no; fi
