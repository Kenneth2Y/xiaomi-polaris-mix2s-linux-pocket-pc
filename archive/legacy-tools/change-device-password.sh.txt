#!/bin/bash
# Passwords are entered directly into passwd over an SSH PTY, never logged.
set -euo pipefail
project=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
exec ssh -F /dev/null -tt -S "$project/build/ssh/control" \
    -o StrictHostKeyChecking=yes \
    -o UserKnownHostsFile=<用户目录>/.ssh/known_hosts \
    -o ConnectTimeout=5 user@172.16.42.1 passwd
