#!/bin/sh
set -eu
export LC_ALL=C PAGER=cat SYSTEMD_PAGER=cat
export http_proxy=http://<本地代理> https_proxy=http://<本地代理>
export HTTP_PROXY="$http_proxy" HTTPS_PROXY="$https_proxy" KERNEL_INSTALL_BYPASS=1
. /usr/share/misc/source_deviceinfo
test "$deviceinfo_flash_kernel_on_update" = false
printf 'Install missing packaged OSK schema (normal signed transaction, mkinitfs bypass):\n'
apk --preserve-env --no-interactive --no-progress --timeout 20 add stevia-schemas
printf '\nRequired schema now present:\n'
gsettings list-schemas | grep '^mobi.phosh.osk$'
printf '\nRestart failed greeter OSK user service without ending the greeter:\n'
uid=$(id -u greetd)
runuser -u greetd -- env XDG_RUNTIME_DIR="/run/user/$uid" DBUS_SESSION_BUS_ADDRESS="unix:path=/run/user/$uid/bus" systemctl --user reset-failed mobi.phosh.OSK.service
runuser -u greetd -- env XDG_RUNTIME_DIR="/run/user/$uid" DBUS_SESSION_BUS_ADDRESS="unix:path=/run/user/$uid/bus" systemctl --user start mobi.phosh.OSK.service
sleep 3
runuser -u greetd -- env XDG_RUNTIME_DIR="/run/user/$uid" DBUS_SESSION_BUS_ADDRESS="unix:path=/run/user/$uid/bus" systemctl --user status mobi.phosh.OSK.service --no-pager -l | head -25
printf '\nActual boot remains verified test05:\n'
dev=$(readlink -f /dev/disk/by-partlabel/boot)
actual=$(head -c 29515776 "$dev" | sha256sum)
printf '%s\n' "$actual"
test "${actual%% *}" = ac2d4496089a3f7658b8cbdae6b6b364ffe81855b42d4434bf3943198eb1cb95
