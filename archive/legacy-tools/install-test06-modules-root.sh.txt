#!/bin/sh
# Add only the new release and an inert-by-default USB experiment policy.
set -eu
umask 077
release=7.1.0-rc1-sdm845-polaris-test06
archive=/tmp/polaris-test06-modules-20261105.tar
destination=/usr/lib/modules/$release
policy=/etc/modprobe.d/polaris-usb-test04.conf
test "$(uname -r)" = 7.1.0-rc1-sdm845-polaris-test05
test ! -e "$destination"
test -f "$policy"
grep -qx "blacklist qcom_pmic_typec_smb2" "$policy"
grep -qx "blacklist qcom_smbx" "$policy"
grep -qx "blacklist qcom_usb_vbus_regulator" "$policy"
df -Pk /usr/lib/modules | awk 'NR == 2 { exit ($4 < 80000) }'
dev=$(readlink -f /dev/disk/by-partlabel/boot)
actual=$(head -c 29515776 "$dev" | sha256sum)
test "${actual%% *}" = ac2d4496089a3f7658b8cbdae6b6b364ffe81855b42d4434bf3943198eb1cb95
stage=$(mktemp -d /usr/lib/modules/.polaris-test06-stage.XXXXXX)
# Copy into a root-owned directory before hashing/extracting the fixed archive.
cp "$archive" "$stage/modules.tar"
printf '%s  %s\n' eac4ccbf674c6ffe88aac49d403d2492cc3fb32740dd19d17f4cae580bbdbeae "$stage/modules.tar" | sha256sum -c -
tar -xf "$stage/modules.tar" -C "$stage"
rm "$stage/modules.tar"
test -f "$stage/usr/lib/modules/$release/modules.dep.bin"
count=$(find "$stage/usr/lib/modules/$release" -name '*.ko.zst' | wc -l)
test "$count" -eq 1156
find "$stage/usr/lib/modules/$release" -type d -exec chmod 755 {} +
mv -T "$stage/usr/lib/modules/$release" "$destination"
rmdir "$stage/usr/lib/modules" "$stage/usr/lib" "$stage/usr" "$stage"
printf 'Installed %s matching modules into independent %s\n' "$count" "$destination"
printf 'Power/Type-C modules remain blacklisted and are not loaded.\n'
printf 'Current kernel remains: '; uname -r
modinfo -k "$release" -F vermagic qcom_pmic_typec_smb2
modprobe -S "$release" -n qcom_pmic_typec_smb2
