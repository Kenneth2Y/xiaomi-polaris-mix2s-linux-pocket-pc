#!/bin/sh
set -eu
export LC_ALL=C PAGER=cat SYSTEMD_PAGER=cat
test "$(uname -r)" = 7.1.0-rc1-sdm845-polaris-test06
test "$(cat /sys/class/power_supply/qcom-battery/capacity)" -ge 20
test "$(cat /sys/class/power_supply/qcom-battery/temp)" -lt 450
test "$(cat /sys/class/usb_role/a600000.usb-role-switch/role)" = host
test "$(systemctl is-enabled polaris-usb-role-test06.service)" = enabled
# Stop only the controller; do not recursively remove its charger/FG suppliers.
sed -i 's|^ExecStop=/sbin/modprobe -r qcom_pmic_typec_smb2$|ExecStop=/sbin/rmmod qcom_pmic_typec_smb2|' /etc/systemd/system/polaris-usb-role-test06.service
systemctl daemon-reload
systemd-analyze verify polaris-usb-role-test06.service
dev=$(readlink -f /dev/disk/by-partlabel/boot)
actual=$(head -c 29515776 "$dev" | sha256sum)
test "${actual%% *}" = e9c7591b9a044f63d3fc4036cee778217ce7fe31a9a31c3cc5205b4977c9adce
printf 'Verified test06 boot and automatic role/GUI services. Hub stays attached for reboot test.\n'
sync
systemctl reboot
