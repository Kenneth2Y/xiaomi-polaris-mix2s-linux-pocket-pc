#!/bin/bash
# Run in the user's own WSL terminal. No tee/logging or secret arguments.
set -euo pipefail
project=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
wifi_ip=$(cat "$project/build/ssh/wifi-address")
python3 -B - "$wifi_ip" <<'PY'
import ipaddress,sys
a=ipaddress.IPv4Address(sys.argv[1])
if not a.is_private or a.is_loopback or a.is_link_local or a in ipaddress.ip_network('172.16.0.0/16'):
    raise SystemExit('Expected the phone private LAN address.')
PY
remote=/tmp/polaris-wifi-save-encrypted-$$.sh
options=(-F /dev/null -S "$project/build/ssh/wifi-control"
    -o StrictHostKeyChecking=yes -o HostKeyAlias=172.16.42.1
    -o UserKnownHostsFile=<用户目录>/.ssh/known_hosts -o ConnectTimeout=5)
ssh "${options[@]}" -o BatchMode=yes "user@$wifi_ip" \
    "umask 077; set -C; cat > $remote" < "$project/tools/wifi-save-encrypted-root.sh"
echo '先输入手机 sudo 密码，随后在 <2.4GHz热点> 提示处输入 Wi-Fi 密码。'
echo '无线密码会在手机本地加密；本命令不记录密码，不会中断当前连接。'
exec ssh "${options[@]}" -tt "user@$wifi_ip" "sudo -- sh $remote"
