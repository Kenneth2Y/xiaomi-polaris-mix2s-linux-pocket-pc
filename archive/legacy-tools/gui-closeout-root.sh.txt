#!/bin/sh
set -eu
export LC_ALL=C PAGER=cat SYSTEMD_PAGER=cat
# Dependencies in vendor units are additive; keep its tty1 conflict and offer tty2.
cat > /etc/systemd/system/greetd.service.d/50-polaris.conf <<'UNIT'
[Service]
ExecStart=
ExecStart=greetd -c /etc/phrog/greetd-config.toml
UNIT
systemctl daemon-reload
systemctl enable --now getty@tty2.service
printf 'Persistent GUI and recovery console:\n'
systemctl is-enabled greetd.service getty@tty2.service polaris-charger-test05.service polaris-wifi-autoconnect.timer
systemctl is-active greetd.service getty@tty2.service polaris-charger-test05.service
printf '\nValidate service definitions:\n'
systemd-analyze verify greetd.service polaris-charger-test05.service
printf '\nUser GUI session:\n'
loginctl list-sessions --no-pager
printf '\nChinese font match:\n'; fc-match ':lang=zh-cn'
printf '\nGUI package versions:\n'
apk --no-network info -v | grep -E '^(phosh-|phoc-|stevia-|greetd-|foot-|epiphany-|nautilus-|font-noto-cjk-|gnome-backgrounds-)'
printf '\nBoot state:\n'
dev=$(readlink -f /dev/disk/by-partlabel/boot)
actual=$(head -c 29515776 "$dev" | sha256sum)
printf 'actual boot prefix: %s\n' "$actual"
test "${actual%% *}" = ac2d4496089a3f7658b8cbdae6b6b364ffe81855b42d4434bf3943198eb1cb95
printf '/boot/boot.img generated by earlier trigger: '; sha256sum /boot/boot.img
printf '\nEffective auto-flash protection:\n'
. /usr/share/misc/source_deviceinfo
test "$deviceinfo_flash_kernel_on_update" = false
printf '%s\n' "$deviceinfo_flash_kernel_on_update"
printf '\nPower:\n'
for f in online status voltage_now current_now current_max; do printf 'charger.%s=' "$f"; cat "/sys/class/power_supply/pmi8998-charger/$f"; done
for f in capacity voltage_now current_now temp; do printf 'battery.%s=' "$f"; cat "/sys/class/power_supply/qcom-battery/$f"; done
printf '\nHardware renderer errors since GUI start:\n'
journalctl -k -b --no-pager | grep -Ei 'gpu.*(fault|hang|timeout)|ringbuffer.*(timeout|fail)|DCS.*(-22|fail)' | tail -12 || true
