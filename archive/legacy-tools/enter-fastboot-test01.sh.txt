#!/bin/bash
# Explicitly authorized 2026-11-04. Verify current boot before graceful reboot.
set -euo pipefail
project=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
tag=${1:-test01-enter-fastboot}
[[ "$tag" =~ ^[a-z0-9-]+$ ]] || exit 2
remote_command=$(cat <<'REMOTE'
sudo -- sh -eu -c 'printf "\nPOLARIS_DMESG_BEGIN\n"; dev=$(readlink -f /dev/disk/by-partlabel/boot); test -b "$dev"; old=$(head -c 30961664 "$dev" | sha256sum); new=$(head -c 29515776 "$dev" | sha256sum); if [ "${old%% *}" != c51af14496f0f9d63117ab1262ad0703a6ce2cde4bcd13b214e0a06cdcde4fc9 ] && [ "${new%% *}" != 68093f62b6aa40abfa5fdd3704d6ab33c97b31ae92993118d1f5ba51b427fd46 ] && [ "${new%% *}" != 6f01281de6087dea502462f373b67daedc0a33403cd41c451753736b25de8922 ]; then printf "Unknown boot image; refusing automatic reboot.\n"; exit 1; fi; printf "Current boot matches a preserved known-working image.\n"; sync; systemctl reboot --reboot-argument=bootloader'
REMOTE
)
ssh -F /dev/null -tt -S "$project/build/ssh/control" \
    -o StrictHostKeyChecking=yes \
    -o UserKnownHostsFile=<用户目录>/.ssh/known_hosts \
    -o ConnectTimeout=5 user@172.16.42.1 "$remote_command" \
    | python3 -B -u "$project/tools/redact-log.py" --auth-until-marker \
    | tee "$project/logs/$tag.log"
