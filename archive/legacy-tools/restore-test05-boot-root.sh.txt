#!/bin/sh
# Restore ONLY the already tested boot image after an unexpected package trigger.
set -eu
export LC_ALL=C PAGER=cat SYSTEMD_PAGER=cat
test "$(uname -r)" = 7.1.0-rc1-sdm845-polaris-test05
stage=/var/lib/polaris-lab/gui-20261104
install -m 600 /tmp/polaris-restore-test05-20261104.img "$stage/boot-test05-restore.img"
img="$stage/boot-test05-restore.img"
test "$(stat -c %s "$img")" = 29515776
expected=ac2d4496089a3f7658b8cbdae6b6b364ffe81855b42d4434bf3943198eb1cb95
actual=$(sha256sum "$img"); test "${actual%% *}" = "$expected"
dev=$(readlink -f /dev/disk/by-partlabel/boot)
test -b "$dev"
test "$(blockdev --getsize64 "$dev")" = 67108864
printf 'Current boot prefix before restore:\n'
head -c 29515776 "$dev" | sha256sum
printf '\nRestore verified test05 image to boot only:\n'
dd if="$img" of="$dev" bs=4M conv=fsync
sync
printf '\nVerify actual restored boot prefix:\n'
actual=$(head -c 29515776 "$dev" | sha256sum)
printf '%s\n' "$actual"
test "${actual%% *}" = "$expected"
printf '\nRunning kernel preserved, no reboot performed:\n'; uname -r
