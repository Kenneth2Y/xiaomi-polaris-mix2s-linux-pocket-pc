#!/bin/bash
# Run interactively in your own WSL terminal; never pass a password as an argument.
set -euo pipefail
project=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
trusted=<用户目录>/.ssh/known_hosts
session="$project/build/ssh"
mkdir -p "$session"
chmod 700 "$session"
test -r "$trusted" || { echo 'Existing Windows known_hosts is unavailable.' >&2; exit 1; }
exec ssh -F /dev/null -M -S "$session/control" -N -f \
    -o ControlPersist=30m -o StrictHostKeyChecking=yes \
    -o UserKnownHostsFile="$trusted" -o ConnectTimeout=5 \
    -o ServerAliveInterval=30 -o ServerAliveCountMax=3 \
    user@172.16.42.1
