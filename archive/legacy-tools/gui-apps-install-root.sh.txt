#!/bin/sh
set -eu
export LC_ALL=C PAGER=cat SYSTEMD_PAGER=cat
export http_proxy=http://<本地代理> https_proxy=http://<本地代理>
export HTTP_PROXY="$http_proxy" HTTPS_PROXY="$https_proxy" KERNEL_INSTALL_BYPASS=1
. /usr/share/misc/source_deviceinfo
test "$deviceinfo_flash_kernel_on_update" = false
printf 'Install essential desktop apps, screenshot verification and Chinese glyph support:\n'
apk --preserve-env --no-interactive --no-progress --timeout 20 add epiphany nautilus grim font-noto-cjk gnome-backgrounds

printf '\nEnable verified graphical login at boot:\n'
systemctl enable greetd.service
printf '\nBoot prefix after package hooks:\n'
dev=$(readlink -f /dev/disk/by-partlabel/boot)
actual=$(head -c 29515776 "$dev" | sha256sum)
printf '%s\n' "$actual"
test "${actual%% *}" = ac2d4496089a3f7658b8cbdae6b6b364ffe81855b42d4434bf3943198eb1cb95
printf '\nAudit package state:\n'
apk --no-network info -v > /var/lib/polaris-lab/gui-20261104/packages.final
cp -p /etc/apk/world /var/lib/polaris-lab/gui-20261104/world.final
printf '\nCharge status:\n'
cat /sys/class/power_supply/qcom-battery/capacity /sys/class/power_supply/pmi8998-charger/current_now
