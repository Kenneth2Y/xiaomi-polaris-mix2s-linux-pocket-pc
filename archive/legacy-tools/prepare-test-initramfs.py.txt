#!/usr/bin/env python3
"""Stage the baseline initramfs module set and its new dependency closure."""
from pathlib import Path
import shutil
import subprocess
from compression import zstd
from bootimg import BootImage, module_names, read_cpio
from candidate_paths import candidate

paths = candidate()
root, source, package = paths["root"], paths["source"], paths["package"]
release = (source / "include/config/kernel.release").read_text().strip()
if release != paths["release"]:
    raise SystemExit(f"Unexpected kernel release: {release}")
modules = package / "usr/lib/modules" / release
# Host kmod understands ARM64 ELF and produces the binary indexes needed by
# the baseline initramfs kmod. The native chroot only has BusyBox depmod.
subprocess.run(["depmod", "-b", str(package), "-m", "/usr/lib/modules", release], check=True)
baseline = BootImage.parse((root / "baseline/boot-known-ssh-20261103.img").read_bytes())
entries = read_cpio(zstd.decompress(baseline.ramdisk))
selected = {n.split("/modules/", 1)[1].split("/", 1)[1] for n in module_names(entries)}
dependencies = {}
for line in (modules / "modules.dep").read_text().splitlines():
    name, deps = line.split(":", 1)
    dependencies[name] = deps.split()
pending = list(selected)
while pending:
    name = pending.pop()
    if name not in dependencies:
        raise SystemExit(f"Baseline module missing in new build: {name}")
    for dependency in dependencies[name]:
        if dependency not in selected:
            selected.add(dependency)
            pending.append(dependency)
stage = paths["stage"]
if stage.exists():
    raise SystemExit("Staging directory already exists; archive it before retrying")
destination = stage / "usr/lib/modules" / release
destination.mkdir(parents=True)
(stage / "lib").symlink_to("usr/lib")
for name in sorted(selected):
    data = (modules / name).read_bytes()
    uncompressed = zstd.decompress(data) if name.endswith(".zst") else data
    if b"vermagic=" + release.encode() + b" " not in uncompressed:
        raise SystemExit(f"Module vermagic mismatch: {name}")
    if not uncompressed.endswith(b"~Module signature appended~\n"):
        raise SystemExit(f"Module signature missing: {name}")
    target = destination / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)
for name in ("modules.builtin", "modules.builtin.modinfo"):
    shutil.copyfile(modules / name, destination / name)
order = [n for n in (modules / "modules.order").read_text().splitlines()
         if n in selected or n + ".zst" in selected]
(destination / "modules.order").write_text("\n".join(order) + "\n")
(root / ("logs/initramfs-module-selection.txt" if paths["profile"] == "test01" else "logs/" + paths["profile"] + "-initramfs-module-selection.txt")).write_text("\n".join(sorted(selected)) + "\n")
subprocess.run(["depmod", "-b", str(stage), "-m", "/usr/lib/modules", release], check=True)
print(f"Staged {len(selected)} signed matching modules for {release}")
print("Dependency and binary indexes generated with host kmod.")
